    public class RelayCommand : ICommand
    {
        private readonly Action<object?> _execute;
        private readonly Func<object?, bool>? _canExecute;

        public RelayCommand(Action<object?> execute, Func<object?, bool>? canExecute = null)
        {
            _execute = execute ?? throw new ArgumentNullException(nameof(execute));
            _canExecute = canExecute;
        }

        public event EventHandler? CanExecuteChanged
        {
            add { CommandManager.RequerySuggested += value; }
            remove { CommandManager.RequerySuggested -= value; }
        }

        public void Execute(object? parameter) => _execute(parameter);

    /// Thread-safe timer-based scheduler for automated screenshot capture
    /// </summary>

    {
        private Timer? _timer;
        private readonly object _lock = new object();
        private bool _disposed;

        /// <summary>
        /// Gets whether the scheduler is currently running
        /// </summary>
        public bool IsRunning

                { 
                    return _timer?.Enabled == true; 
                } 
            } 
        }

        /// <summary>
        /// Gets the current interval in milliseconds, or 0 if not running
        /// </summary>

        {
            get
            {
                lock (_lock)
                {
                    return _timer?.Interval ?? 0;
                }
            }
        }

        /// <summary>
        /// Starts the scheduler with the specified interval and action
        /// Stops any existing schedule first
        /// </summary>
        /// <param name="intervalMs">Interval between executions in milliseconds (minimum 100)</param>
        /// <param name="action">Action to execute on each timer tick</param>
        /// <exception cref="ArgumentOutOfRangeException">Thrown when interval is less than 100ms</exception>
        /// <exception cref="ArgumentNullException">Thrown when action is null</exception>
        /// <exception cref="ObjectDisposedException">Thrown when scheduler has been disposed</exception>
        public void Start(int intervalMs, Action action)
        {
            if (_disposed)
                throw new ObjectDisposedException(nameof(OcrScheduler));

            if (intervalMs < 100)
                throw new ArgumentOutOfRangeException(nameof(intervalMs), "Interval must be at least 100 milliseconds");

            if (action is null)
                throw new ArgumentNullException(nameof(action));

            lock (_lock)
            {
                // Stop any existing timer
                StopInternal();

                // Create and configure new timer
                _timer = new Timer(intervalMs);
                _timer.Elapsed += (sender, e) =>
                {
                    try
                    {
                        action();
                    }
                    catch
                    {
                        // Swallow exceptions to prevent timer from stopping
                        // Logging should be handled by the action itself
                    }
                };
                
                _timer.AutoReset = true;
                _timer.Start();
            }
        }

        /// <summary>
        /// Stops the current schedule if running
        /// Safe to call multiple times or when not running
        /// </summary>
        public void Stop()
        {
            if (_disposed) return;

            lock (_lock)
            {
                StopInternal();
            }
        }

        /// <summary>
        /// Changes the interval of a running scheduler
        /// No effect if scheduler is not currently running
        /// </summary>
        /// <param name="intervalMs">New interval in milliseconds (minimum 100)</param>
        /// <exception cref="ArgumentOutOfRangeException">Thrown when interval is less than 100ms</exception>
        public void ChangeInterval(int intervalMs)
        {

                throw new ArgumentOutOfRangeException(nameof(intervalMs), "Interval must be at least 100 milliseconds");

            lock (_lock)

                if (_timer != null && _timer.Enabled)

                }

            lock (_lock)
            {
                StopInternal();

        }
